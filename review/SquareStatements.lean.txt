-- Introduction: formal circuit correctness implies upstream input correctness.
theorem formalCorrect_implies_correct
    {F : Type*} [Field F] {a b c : ℕ}
    (P : OAI.MatrixMultiplication.Arithmetic.MatrixAlgorithm F a b c) (hP : FormalCorrect P)
    : P.Correct

-- Appendix A: both interfaces denote exactly the same formal-polynomial exponent.
-- This does NOT assert equality with upstream functional omega over finite fields.
theorem formalOmega_eq_exactRankExponent
    (F : Type*) [Field F]
    : formalOmega F = AuxiliarySeparation.exactRankExponent F

-- Appendix A: the extension is the algebraic closure of this particular field.
theorem formalOmega_eq_algebraicClosure
    (F : Type*) [Field F]
    : formalOmega F = formalOmega (AlgebraicClosure F)

-- Appendix A: fixed coefficient extension before taking all tensor powers.
theorem scalar_descent
    {F X Y Z : Type*} [Field F] [Fintype X] [Fintype Y] [Fintype Z]
    (T : OAI.MatrixMultiplication.Foundation.Tensor F X Y Z) (r : ℕ)
    (h : OAI.MatrixMultiplication.Foundation.Tensor.RankAtMost
      (fun x y z => algebraMap F (AlgebraicClosure F) (T x y z)) r)
    : ∃ q : ℕ, 1 ≤ q ∧ ∀ m : ℕ, 1 ≤ m →
        OAI.MatrixMultiplication.Foundation.Tensor.RankAtMost
          (OAI.MatrixMultiplication.Foundation.Tensor.power T m) (q ^ 2 * r ^ m)

-- Lemma 2.1: the catalyst is returned unchanged and is nonzero.
theorem asymptotic_cancellation
    {K : Type*} [Field K] [IsAlgClosed K]
    (A B Z : FiniteTensor K) (hZ : Nonzero Z)
    (h : AsymptoticallyRestricts (product A Z) (product B Z))
    : AsymptoticallyRestricts A B

-- Full Lemma 2.3: monotonicity, exact matrix scaling, and bounded copies.
theorem value_matrix_factor
    {K : Type*} [Field K] [IsAlgClosed K]
    (T : NonzeroTensor K)
    (hcyclic : Isomorphic T.val (permute .yzx T.val))
    : (∀ U : NonzeroTensor K, Isomorphic U.val (permute .yzx U.val) →
        AsymptoticallyRestricts T.val U.val → valueAtOmega U ≤ valueAtOmega T) ∧
      (∀ d : ℕ, ∀ hd : 1 ≤ d,
        valueAtOmega (T.product (nonzeroSquare d hd)) =
          (d : ℝ) ^ formalOmega K * valueAtOmega T) ∧
      (∀ b : ℕ, ∀ hb : 1 ≤ b,
        valueAtOmega (T.copies b hb) ≤ (b : ℝ) ^ formalOmega K * valueAtOmega T)

-- Proposition 2.4: index an unbounded sequence of scales by j → n j.
-- There is no requirement that the p_i sum to one.
-- Permutations may depend on both the constituent and the scale.
theorem general_extraction
    {K : Type*} [Field K] [IsAlgClosed K]
    {ι : Type} [Fintype ι]
    (A : NonzeroTensor K) (B : ι → NonzeroTensor K)
    (n a b u h : ℕ → ℕ) (m : ι → ℕ → ℕ)
    (π : ι → ℕ → LegPermutation)
    (hn : ∀ j, 1 ≤ n j) (ha : ∀ j, 1 ≤ a j)
    (hb : ∀ j, 1 ≤ b j) (hu : ∀ j, 1 ≤ u j) (hh : ∀ j, 1 ≤ h j)
    (hn_tendsto : Filter.Tendsto n Filter.atTop Filter.atTop)
    (hconvert : ∀ j, AsymptoticallyRestricts
      (copies (u j) (product (power (symmetrize A.val) (n j)) (square (a j))))
      (copies (h j) (product (square (b j))
        (tensorProductList (List.ofFn (fun l : Fin (Fintype.card ι) =>
          let i := (Fintype.equivFin ι).symm l
          power (symmetrize (permute (K := K) (π i j) (B i).val)) (m i j)))))))
    (p : ι → ℝ) (α β : ℝ)
    (hu_rate : Filter.Tendsto (fun j => Real.log (u j : ℝ) / (n j : ℝ))
      Filter.atTop (nhds 0))
    (hm_rate : ∀ i, Filter.Tendsto (fun j => (m i j : ℝ) / (n j : ℝ))
      Filter.atTop (nhds (p i)))
    (hh_rate : Filter.Tendsto (fun j => Real.log (h j : ℝ) / (3 * (n j : ℝ)))
      Filter.atTop (nhds α))
    (hab_rate : Filter.Tendsto
      (fun j => Real.log ((b j : ℝ) / (a j : ℝ)) / (3 * (n j : ℝ)))
      Filter.atTop (nhds β))
    : Real.log (valueAtOmega A) ≥
        (∑ i, p i * Real.log (valueAtOmega (B i))) + α + formalOmega K * β

-- Proposition 2.7: local inequality; individual block maps need not agree.
theorem shared_leg_value_inequality
    {K : Type*} [Field K] [IsAlgClosed K]
    {ι : Type} [Fintype ι]
    (A : NonzeroTensor K) (B : ι → NonzeroTensor K)
    (D : MatrixShape) (E : ι → MatrixShape)
    (identity : SharedLegIdentity A.val (fun i => (B i).val) D E)
    : (D.volume : ℝ) * normalizedValueAtOmega A ≥
        ∑ i, ((E i).volume : ℝ) * normalizedValueAtOmega (B i)

-- Theorem 1.1: the source bound is border rank, not a hidden value inequality.
-- c and m are expanded literally from matrix dimensions and branch labels.
theorem identities_to_exponent
    {F : Type*} [Field F]
    {V E : Type} [Fintype V] [DecidableEq V] [Fintype E]
    (A : V → FiniteTensor (AlgebraicClosure F)) (hA : ∀ v, Nonzero (A v))
    (q : E → ℕ) (σ : E → V) (w : (e : E) → Fin (q e) → V)
    (D : E → MatrixShape) (Dout : (e : E) → Fin (q e) → MatrixShape)
    (identities : ∀ e, SharedLegIdentity (A (σ e)) (fun i => A (w e i)) (D e) (Dout e))
    (s t : V) (hst : s ≠ t) (hterminal : ∀ e, σ e ≠ t)
    (d r : ℕ) (hd : 2 ≤ d) (hr : 1 ≤ r)
    (ht : Isomorphic (symmetrize (A t)) (square d))
    (hs : BorderRankAtMost (A s) r)
    (μ : E → ℝ) (hμ : ∀ e, 0 ≤ μ e)
    (hcertificate : ∀ v, v ≠ t →
      (∑ e, if σ e = v then ((D e).volume : ℝ) * μ e else 0) -
      (∑ e, (∑ i : Fin (q e), if w e i = v then ((Dout e i).volume : ℝ) else 0) * μ e)
      ≤ if v = s then 1 else 0)
    (hL : (d : ℝ)⁻¹ <
      ∑ e, (∑ i : Fin (q e), if w e i = t then ((Dout e i).volume : ℝ) else 0) * μ e)
    : formalOmega F ≤ 3 * Real.log (r : ℝ) /
      (Real.log (d : ℝ) + Real.log
        (∑ e, (∑ i : Fin (q e), if w e i = t then ((Dout e i).volume : ℝ) else 0) * μ e))

-- Proposition 3.1: canonical sums retain both exact constituent tensors.
theorem convolution_neighbor_degeneration
    {K : Type*} [Field K] (r b : ℕ+) (hb : 2 ≤ (b : ℕ))
    : Degenerates (product (convolution (K := K) r b) dotShape.tensor)
        (neighborSplit r b hb)

-- Proposition 3.1: the shared target retains the middle first-leg reversal.
theorem convolution_tripling_degeneration
    {K : Type*} [Field K] (r : ℕ+) (hr : 2 ≤ (r : ℕ))
    : Degenerates
        (convolution (K := K) r ⟨4 * (r : ℕ) - 1, by omega⟩)
        (triplingSplit r hr)

-- Only this individual block isomorphism removes the reversed first coordinate.
theorem tripling_middle_isomorphism
    {K : Type*} [Field K] (r : ℕ+)
    : Isomorphic (middleConvolution (K := K) r)
        (permute .xzy (convolution r r))

-- Proposition 3.1 lower boundary, with its exact input transposition.
theorem convolution_boundary_isomorphism
    {K : Type*} [Field K] (r : ℕ+) (hr : 2 ≤ (r : ℕ))
    : Isomorphic
        (convolution (K := K) r ⟨(r : ℕ) - 1, by omega⟩)
        (permute .yxz (convolution ⟨(r : ℕ) - 1, by omega⟩ r))

-- Explicit certificate's scalar terminal value, used in the finite square bound.
-- The closed product is written inline; no assumed LP optimum is used.
theorem finite_convolution_bound
    (F : Type*) [Field F] (a : ℕ) (ha : 2 ≤ a)
    : formalOmega F ≤ 3 * Real.log (2 * (a : ℝ) - 1) /
      (Real.log 2 + Real.log (((3 * (a : ℝ) - 1) / 4) *
        ∏ j ∈ Finset.Icc 1 (a - 1), (1 + 1 / (3 * (j : ℝ)))))

-- Main theorem: no characteristic, infinitude, perfectness or closure hypothesis.
theorem formalOmega_le_nine_quarters
    (F : Type*) [Field F]
    : formalOmega F ≤ (9 : ℝ) / 4

-- Strong public conclusion: one constant for every positive size, polynomial identity.
theorem formal_matrix_multiplication_cost_le
    (F : Type*) [Field F] (ε : ℝ) (hε : 0 < ε)
    : ∃ C : ℝ, 0 < C ∧ ∀ n : ℕ, 1 ≤ n →
        ∃ P : OAI.MatrixMultiplication.Arithmetic.MatrixAlgorithm F n n n,
          FormalCorrect P ∧ (P.cost : ℝ) ≤ C * (n : ℝ) ^ ((9 : ℝ) / 4 + ε)

-- Compatibility corollary in OpenAI/Navot's unchanged functional model.
theorem omega_le_nine_quarters
    (F : Type*) [Field F]
    : OAI.MatrixMultiplication.Arithmetic.omega F ≤ (9 : ℝ) / 4
